cmake_minimum_required(VERSION 3.22)
project(ReelGlue VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.15 GIT_SHALLOW TRUE)
FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.5.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE Catch2)
juce_add_plugin(ReelGlue
    COMPANY_NAME "ReelGlue Audio" BUNDLE_ID "audio.reelglue.plugin" PLUGIN_MANUFACTURER_CODE Rglu PLUGIN_CODE RgL1
    FORMATS VST3 Standalone PRODUCT_NAME "ATR-102"
    VST3_CATEGORIES Fx Dynamics IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(ReelGlue)
target_sources(ReelGlue PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(ReelGlue PUBLIC Source)
target_compile_definitions(ReelGlue PUBLIC JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ReelGlue PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra juce::juce_recommended_config_flags)
if(MSVC)
    target_compile_options(ReelGlue PRIVATE /W4 /permissive- /fp:precise)
else()
    target_compile_options(ReelGlue PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(ReelGlueTests Tests/DspTests.cpp)
target_include_directories(ReelGlueTests PRIVATE Source)
target_link_libraries(ReelGlueTests PRIVATE Catch2::Catch2WithMain)
add_test(NAME ReelGlueDSP COMMAND ReelGlueTests)
option(REELGLUE_BUILD_RENDERER "Build UI snapshot and processor integration tool" OFF)
if(REELGLUE_BUILD_RENDERER)
    add_executable(ReelGlueRender)
    set_target_properties(ReelGlueRender PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/ReelGlueRender_artefacts/$<CONFIG>")
    target_sources(ReelGlueRender PRIVATE Tests/Render.cpp)
    target_include_directories(ReelGlueRender PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ReelGlue_artefacts/JuceLibraryCode" "${juce_SOURCE_DIR}/modules")
    target_compile_definitions(ReelGlueRender PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    target_link_libraries(ReelGlueRender PRIVATE ReelGlue)
endif()
